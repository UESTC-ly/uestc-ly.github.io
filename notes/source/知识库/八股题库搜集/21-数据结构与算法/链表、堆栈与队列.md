```python
class Node(object):
    def __init__(self,item,next_=None):
        self.item = item
        self.next = next_

class Solution(object):
    def mergeTwoLists(self, l1,l2):     
        head = Node(0)
        cur = head
        while l1 != None and l2 != None:
            if l1.item <= l2.item:
                cur.next = l1
                l1 =  l1.next
            else:
                cur.next = l2
                l2 = l2.next
            cur = cur.next
        if l1 != None:
            cur.next = l1
        if l2 != None:
            cur.next = l2
        return head.next    
```

## 删除无序链表中的重复项

给定一个无序的链表，去掉其重复项，并保留原顺序，例如链表1->3->1->5->5->7，去掉重复项后为1->3->5->7
```python
class ListNode:
    def __init__(self, x):
        self.val = x
        self.next = None
class Solution:
    def deleteDuplicates(self, head: ListNode) -> ListNode:
        if head is None:
            return head
        outer = head
        while outer:
            inpre = outer
            inner = inpre.next
            while inner:
                if inner.val == outer.val:
                    inner = inner.next
                    inpre.next = inner
                else:
                    inpre = inner
                    inner = inner.next
            outer = outer.next
        return head        
```

## 单调栈和单调队列解决什么问题？

单调结构用于在 $O(n)$ 时间内处理“最近更大/更小元素”“滑动窗口最值”等问题。

核心思想：

- 栈或队列中元素保持单调。
- 新元素进入时，弹出不可能再成为答案的旧元素。
- 每个元素最多入栈/入队一次、出栈/出队一次，所以总复杂度 $O(n)$。

## 单调栈如何在 $O(n)$ 内找到下一个更大元素？

问题：对数组中每个元素，找到右边第一个比它大的元素。

模板：

```python
def next_greater(nums):
    ans = [-1] * len(nums)
    stack = []  # 存下标，保持对应值单调递减
    for i, x in enumerate(nums):
        while stack and nums[stack[-1]] < x:
            ans[stack.pop()] = x
        stack.append(i)
    return ans
```

为什么是 $O(n)$？

- 每个元素最多入栈一次。
- 每个元素最多出栈一次。

AI 场景类比：

- 在时序特征中找未来第一个超过阈值的点。
- 在显存使用曲线中找下一次峰值。
- 在日志序列中找下一次更高严重级别事件。

## 单链表归并排序

> 题目: 单链表的归并排序。
> 公司: 百度

```cpp
void MergeSort(Node **head_ref) {
    Node *head  = *head_ref;
    Node *left;
    Node *right;

    // 判断是否是null
    if(head == NULL || head->next == NULL) {
        return;
    }

    // 链表分成两个部分，left 和 right
    split(head, &left, &right);

    MergeSort(left);
    MergeSort(right);

    *head_ref = Merge(left, right);
}

// 左右各一半，
void split(Node *head, Node **left, Node **right) {
    //1. 先计算长度n，分别选择前一半和后一半。
    //2. 使用快慢指针，各取一半
    int n = 0;
    Node *cur = head;
    while(cur != NULL) {
        n ++;
    }
    *left = head;

    int k = n / 2;
    cur = head;
    Node *p = NULL;
    while(k--) {
        p = cur;
        cur = cur->next;
    }
    p->next = NULL;

    *right = cur;
}

Node* Merge(Node *left, Node *right) {
    // merge right to left
    Node *head = NULL;
    head->data = -1;
    Node *p = head;    

    while(left != NULL && right != NULL) {
        if(left->data <= right->data) {
            p->next = left;
            left = left->next;
        }
        else {
            p->next = right;
            right = right->next;
        }
        p = p->next;
    }

    if(left != NULL) {
        p->next = left;
    }
    if(right != NULL) {
        p->next = right;
    }

    return head->next;
}
```

## 单链表插入排序

> 题目: 单链表的插入排序（升序）。
> 公司: 百度

```cpp
struct Node {
    int data;
    struct Node * next;
};

void InsertLinked(Node** sorted, Node* tmp) {
    Node* cur;
    // 当前插入节点是最小的值
    if(*sorted == NULL || tmp->data <= (*sorted)->data) {
        tmp->next = *sorted;
        *sorted = tmp;
    }
    else { // 找到插入的位置
        cur = *sorted;
        while(cur->next != NULL && tmp->data > cur->next->data) {
            cur = cur->next;
        }
        tmp->next = cur->next;
        cur->next = tmp;
    }
}

void InsertSort(Node** head) {
    // 有序链表
    Node *sorted = NULL;
    Node * cur = *head;

    while(cur != NULL) {
        Node *next = cur->next;

        // 将cur插入到sorted中，这是一个有序的链表
        InsertLinked(&sorted, cur);

        cur = next;
    }
    *head = sorted;
}

```

## 反转链表（206）

(循环算法，递归算法)  微软

~~~
输入: 1->2->3->4->5->NULL 
输出: 5->4->3->2->1->NULL
~~~

```python
class ListNode(object):
    def __init__(self, x):
        self.val = x
        self.next = None

    def __repr__(self):
        if self:
            return "{} -> {}".format(self.val, repr(self.next))

class Solution(object):
    '''循环算法'''
    def reverseList(self, head):
        dummy = ListNode(float("-inf"))
        while head:
            dummy.next, head.next, head = head, dummy.next, head.next
        return dummy.next        

class Solution2(object):
    '''递归算法'''
    def reverseList(self, head):
        [begin, end] = self.reverseListRecu(head)
        return begin

    def reverseListRecu(self, head):
        if not head:
            return [None, None]

        [begin, end] = self.reverseListRecu(head.next)

        if end:
            end.next = head
            head.next = None
            return [begin, head]
        else:
            return [head, head]
```

## 在双向链表中删除指定元素（微软）

```python
class Node(object):
    '''双向节点'''
    def __init__(self, item):
        self.item = item
        self.next = None
        self.prev = None

class DLinkedlist(object):
    def __init__(self):
        self._head = None

    def remove(self, item):
        """删除元素"""
        if self.is_empty():
            return
        else:
            cur = self._head
            if cur.item == item:
# 如果首节点的元素即是要删除的元素
                if cur.next == None:
# 如果链表只有这一个节点
                    self._head = None
                else:
# 将第二个节点的prev设置为None
                    cur.next.prev = None
# 将_head指向第二个节点
                    self._head = cur.next
                return

            while cur != None:
                if cur.item == item:
# 将cur的前一个节点的next指向cur的后一个节点
                    cur.prev.next = cur.next
# 将cur的后一个节点的prev指向cur的前一个节点
                    cur.next.prev = cur.prev
                    break
                cur = cur.next

```

## 堆为什么适合维护最值和 Top K？

堆是一种特殊的完全二叉树，满足堆序性质：

- 大顶堆：父节点值大于或等于子节点值。
- 小顶堆：父节点值小于或等于子节点值。

堆的优势：

- 插入元素：$O(\log n)$。
- 删除堆顶：$O(\log n)$。
- 查看最值：$O(1)$。
- 建堆：$O(n)$。

堆特别适合“只关心最大或最小的少量元素”的问题。例如：

- Top K 文档召回。
- Top K token 采样。
- Beam Search 保留得分最高的候选序列。
- 推理请求按优先级调度。
- 多路归并合并多个有序流。

## 堆排序、优先队列和 Beam Search 有什么关系？

堆排序是利用堆结构反复取最值来完成排序，复杂度为 $O(n\log n)$，空间复杂度可做到 $O(1)$，但不稳定。

优先队列是堆最常见的抽象接口：

- `push`：加入候选。
- `pop`：取出优先级最高或最低的元素。
- `peek`：查看当前最优元素。

Beam Search 是序列生成中的典型优先队列场景：

1. 每一步扩展当前 beam 中的候选序列。
2. 为新候选计算累计得分。
3. 从所有候选中保留 Top B 条。
4. 重复直到结束。

如果候选总数很大，用完整排序代价高；使用堆或 Top K 选择可以减少开销。

注意：真实大模型推理中，Top K / Top P 通常由高度优化的 GPU kernel 完成，但它背后的算法思想仍然是避免对整个 vocabulary 做完整排序。

## 堆排序和优先队列在 Top K 中如何使用？

堆是一种完全二叉树，通常用数组实现。

下标关系：

- 父节点：`(i - 1) // 2`。
- 左孩子：`2 * i + 1`。
- 右孩子：`2 * i + 2`。

Top K 场景：

- 求最大的 K 个元素：维护大小为 K 的小顶堆。
- 遍历新元素，如果大于堆顶，则替换堆顶并下沉。
- 总复杂度 $O(n\log K)$。

大模型场景：

- logits 中选 Top K token。
- RAG 检索结果取 Top K 文档。
- Beam Search 保留 Top K 序列。
- 调度系统按优先级处理请求。

注意：

- 如果 K 很小，堆非常合适。
- 如果需要完整排序，用堆不一定最快。
- GPU 上 Top K 会有专门 kernel，思想仍然是避免全量排序。

## 堆排序法

这里说的是堆排序法，而不是快排或者希尔排序。虽然理论时间复杂度都是 `O(nlogn)`，但是堆排在做 topK 的时候有一个优势，就是可以维护一个仅包含 k 个数字的小顶堆（想清楚，为啥是小顶堆哦），当新加入的数字大于堆顶数字的时候，将堆顶元素剔除，并加入新的数字。

用 C++ 来说明，堆在 stl 中是 priority_queue（不是 set）。

```cpp
int main() {
    const int topK = 3;
    vector<int> vec = {4,1,5,8,7,2,3,0,6,9};
    priority_queue<int, vector<int>, greater<>> pq;    // 小顶堆
    for (const auto& x : vec) {
        pq.push(x);
        if (pq.size() > topK) {
            // 如果超出个数，则弹出堆顶（最小的）数据
            pq.pop();
        }
    }

    while (!pq.empty()) {
        cout << pq.top() << endl;    // 输出依次为7,8,9
        pq.pop();
    }

    return 0;
}
```

## 堆的插入、删除和建堆过程是什么？

堆的核心操作是上浮和下沉。

插入元素：

1. 将新元素放到数组末尾。
2. 与父节点比较。
3. 如果违反堆序，则交换并继续上浮。

删除堆顶：

1. 将堆顶和末尾元素交换。
2. 删除末尾元素。
3. 从根节点开始下沉，直到恢复堆序。

建堆：

1. 从最后一个非叶子节点开始。
2. 依次向前执行下沉。
3. 总复杂度是 $O(n)$，不是 $O(n\log n)$。

Python 小顶堆示例：

```python
import heapq

def top_k(nums, k):
    heap = []
    for x in nums:
        if len(heap) < k:
            heapq.heappush(heap, x)
        elif x > heap[0]:
            heapq.heapreplace(heap, x)
    return sorted(heap, reverse=True)
```

AI 面试中可结合：

- 采样前只保留概率最高的 K 个 token。
- 多路召回时合并不同检索器返回的候选。
- GPU 任务队列按 deadline、cost、priority 排序。

## 如何用三种语言实现 Top K 小顶堆？

求最大的 K 个元素时，维护大小为 K 的小顶堆。

Python：

```python
import heapq

def top_k_largest(nums, k):
    heap = []
    for x in nums:
        if len(heap) < k:
            heapq.heappush(heap, x)
        elif x > heap[0]:
            heapq.heapreplace(heap, x)
    return sorted(heap, reverse=True)
```

C/C++：

```cpp
#include <functional>
#include <queue>
#include <vector>
using namespace std;

vector<int> topKLargest(vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int x : nums) {
        if ((int)pq.size() < k) pq.push(x);
        else if (x > pq.top()) {
            pq.pop();
            pq.push(x);
        }
    }
    vector<int> ans;
    while (!pq.empty()) {
        ans.push_back(pq.top());
        pq.pop();
    }
    return ans;
}
```

Java：

```java
import java.util.*;

class TopK {
    List<Integer> topKLargest(int[] nums, int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int x : nums) {
            if (pq.size() < k) pq.offer(x);
            else if (x > pq.peek()) {
                pq.poll();
                pq.offer(x);
            }
        }
        return new ArrayList<>(pq);
    }
}
```

## 如何用三种语言实现链表中点？

核心思想：快慢指针。慢指针每次走一步，快指针每次走两步，快指针到尾部时慢指针在中点。

AI 场景类比：流式缓冲区、token 序列截断和滑动窗口都可以借鉴“双指针不同速度扫描”的思想。

Python：

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def middle_node(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    return slow
```

C/C++：

```cpp
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int v) : val(v), next(nullptr) {}

        slow = slow->next;
```

## 如何用哈希表 + 双向链表实现 LRU Cache？

LRU Cache 要支持：

- `get(key)`：查找并把节点移到最近使用位置。
- `put(key, value)`：插入或更新，并在超容量时淘汰最久未使用节点。

单独链表的问题：查找慢，$O(n)$。

单独哈希表的问题：不知道谁最久未使用。

组合方案：

```text
HashMap: key -> Node
DoubleLinkedList: 按访问时间维护节点顺序
```

复杂度：

- 查询：$O(1)$。
- 插入：$O(1)$。
- 删除：$O(1)$。
- 移动到头部：$O(1)$。

AI 场景：

- RAG query cache。
- embedding cache。
- 多轮对话 session cache。
- 推理服务中的 prefix / KV cache 淘汰。
- 特征工程中热门特征缓存。

面试中可以强调：LRU 是数据结构组合题，不是单一结构题；哈希表负责定位，双向链表负责顺序。

## 归并排序、快速排序、堆排序如何对比？

三者都是高频排序算法：

| 算法 | 平均时间 | 最坏时间 | 空间 | 稳定性 | 核心思想 |
| --- | --- | --- | --- | --- | --- |
| 归并排序 | $O(n\log n)$ | $O(n\log n)$ | $O(n)$ | 稳定 | 分治 + 合并 |
| 快速排序 | $O(n\log n)$ | $O(n^2)$ | $O(\log n)$ | 不稳定 | pivot + 分区 |
| 堆排序 | $O(n\log n)$ | $O(n\log n)$ | $O(1)$ | 不稳定 | 堆结构维护最值 |

选择建议：

- 需要稳定性：归并排序。
- 通用内存排序：优化后的快排 / introsort。
- 需要 Top K 或优先队列：堆。
- 数据太大无法一次进内存：外部归并排序。

## 快慢指针为什么能求链表中点和判断环？

快慢指针是链表高频技巧：

- 慢指针每次走 1 步。
- 快指针每次走 2 步。

求中点：

```python
def middle_node(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    return slow
```

判断环：

- 如果链表有环，快指针最终会追上慢指针。
- 如果无环，快指针会先到 `None`。

应用：

- 链表中点。
- 环检测。
- 删除倒数第 k 个节点。
- 判断回文链表。

AI 工程类比：

- 双指针思想常用于滑动窗口、token 序列截断、流式缓冲区处理。
- 快慢指针本质是用不同速度扫描来发现结构性质。

## 手撕】手写链表两两交换节点，注意别把指针弄断，并说清空链表、单节点这两个边界。

**回答：**

题目要求交换节点连接关系，不是只交换节点值。使用哑节点 dummy 统一处理头节点变更；prev 始终指向下一组待交换节点的前驱。若当前结构是 `prev -> first -> second -> rest`，保存 rest 后依次改成 `prev -> second -> first -> rest`，最后让 prev 移到 first。

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def swap_pairs(head):
    dummy = ListNode(next=head)
    prev = dummy
    while prev.next is not None and prev.next.next is not None:
        first = prev.next
        second = first.next
        rest = second.next

        first.next = rest
        second.next = first
        prev.next = second
        prev = first
    return dummy.next
```

循环不变量是：prev 之前的所有完整二元组已经交换，prev.next 开始的后缀仍保持原顺序。每轮只修改当前两个节点及前驱的连接，不创建替代节点，也不丢失剩余后缀；循环结束时后缀不足两个节点，因此无需交换。

输入 `1 -> 2 -> 3 -> 4 -> 5`，输出 `2 -> 1 -> 4 -> 3 -> 5`。空链表返回 `None`；单节点直接返回原节点；奇数长度最后一个节点保留；重复值不影响正确性，因为交换的是节点身份。假设输入是有限无环单链表，若可能有环，需先检测并定义处理策略。

时间复杂度 O(n)，额外空间 O(1)。验证时除了比较值序列，还要确认节点对象一个不少、没有新环、最后一个奇数节点未变，且连续交换两次可以恢复原始节点顺序。这样才能防止“值看起来对了，但指针结构已经错了”的隐藏错误。

## 手撕代码：实现反转链表。

**回答：**

反转链表的核心是逐步改变每个节点的 `next` 指针。需要维护三个指针：`prev` 表示已经反转部分的头节点，`cur` 表示当前处理节点，`nxt` 暂存原链表的下一个节点，避免断链。

Python 实现如下：

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


def reverse_list(head: ListNode | None) -> ListNode | None:
    prev = None
    cur = head
    while cur:
        nxt = cur.next
        cur.next = prev
        prev = cur
        cur = nxt
    return prev
```

Go 实现如下：

```go
type ListNode struct {
    Val  int
    Next *ListNode
}

func reverseList(head *ListNode) *ListNode {
    var prev *ListNode
    cur := head
    for cur != nil {
        next := cur.Next
        cur.Next = prev
        prev = cur
        cur = next
    }
    return prev
}
```

时间复杂度是 O(n)，每个节点只遍历一次；空间复杂度是 O(1)，只使用常数个指针。边界情况包括空链表、单节点链表、两节点链表，都可以被这套逻辑自然覆盖。

## 手撕：反转链表

答：反转链表最常见的是迭代三指针写法，时间复杂度 $O(n)$，空间复杂度 $O(1)$。

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def reverse_list(head: ListNode | None) -> ListNode | None:
    prev = None
    cur = head

    while cur is not None:
        nxt = cur.next
        cur.next = prev
        prev = cur
        cur = nxt

    return prev
```

递归版本也可以写：

```python
def reverse_list_recursive(head: ListNode | None) -> ListNode | None:
    if head is None or head.next is None:
        return head

    new_head = reverse_list_recursive(head.next)
    head.next.next = head
    head.next = None
    return new_head
```

面试讲解时重点说清楚：`nxt` 保存原链表的后续节点，`cur.next = prev` 完成指针反转，然后 `prev` 和 `cur` 同步前进。最后 `prev` 指向新头节点。

## 求单链表的中间节点

快、慢指针实现
```python
def find_middle_node(head):
    slow, fast = head, head
    fast = fast.next if fast else None
```

## 环形链表（141）

~~~
给定一个链表，判断链表中是否有环, 并找到第一个相交的点
思路：设置两个指针slow和fast，一个步长为1，一个步长为2进行遍历。如果有环，则slow和fast总会在某一点相遇。如果没有环，则fast会先为空，或者fast.next为空
~~~

```python
class Solution(object):
    def hasCycle(self, head, firstMeetNode):
        fast = slow = head
        while fast and fast.next:
            fast = fast.next.next
            slow = slow.next
            if slow == fast: # 有环
                fast = head
                while fast != slow:
                    fast = fast.next
                    slow = slow.next
                firstMeetNode = fast
                return True
        return False
```

> **解析**: 

    dp[i][j]: 表示[i,j]区间上最小值.
    dp[i][j] = min(dp[i][j], dp[i][k]+dp[k+1][j]+p[i-1]*p[k]*p[l]) k in [i,j-1]

```
int MatrixChainOrder(int p[], int n) {    
    int dp[n][n]; 
    int i, j, k, L, q;   
    for (i=1; i<n; i++) { 
        dp[i][i] = 0; 
    }
    // L is chain length. 
    for (L=2; L<n; L++) { 
        for (i=1; i<n-L+1; i++) { 
            j = i+L-1; 
            dp[i][j] = INT_MAX; 
            for (k=i; k<=j-1; k++) { 
                q = dp[i][k] + dp[k+1][j] + p[i-1]*p[k]*p[j]; 
                if (q < dp[i][j]) dp[i][j] = q; 
            } 
        } 
    } 
    return m[1][n-1]; 
}
```

## 算法题：Leetcode 143. 重排链表

**回答：**

这题要求原地重排：

$$
L_0\rightarrow L_1\rightarrow\cdots\rightarrow L_n
$$

变为：

$$
L_0\rightarrow L_n\rightarrow L_1\rightarrow L_{n-1}\rightarrow\cdots
$$

标准解法分三步：快慢指针找到前半段尾节点；反转后半段；把两段链表交替合并。Python 实现如下：

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


def reorder_list(head: ListNode | None) -> None:
    if head is None or head.next is None:
        return

    slow = fast = head
    while fast.next is not None and fast.next.next is not None:
        slow = slow.next
        fast = fast.next.next

    second = slow.next
    slow.next = None

    prev = None
    while second is not None:
        nxt = second.next
        second.next = prev
        prev = second
        second = nxt

    first, second = head, prev
    while second is not None:
        next_first = first.next
        next_second = second.next
        first.next = second
        second.next = next_first
        first = next_first
        second = next_second
```

找中点、反转和合并各遍历一次链表，时间复杂度为 $O(n)$，额外空间复杂度为 $O(1)$。使用数组加双指针也能做到 $O(n)$ 时间，但需要 $O(n)$ 额外空间，没有体现题目希望考察的链表断链、反转和交错合并能力。

边界测试至少包括空链表、单节点、两个节点、奇数长度和偶数长度。尤其要先执行 `slow.next = None` 断开前后半段，否则合并时可能形成环。

## 链表的核心思想是什么？

链表通过指针或引用将节点串联起来。每个节点通常包含数据和指向下一个节点的指针。

常见类型：

- 单链表。
- 双向链表。
- 循环链表。

优势：

- 插入和删除节点只需修改指针，已知节点时为 $O(1)$。
- 不要求连续内存。
- 适合频繁移动节点的场景。

局限：

- 不支持 $O(1)$ 随机访问。
- 需要额外指针空间。
- 缓存局部性差。
- 代码更容易出现空指针、断链和循环错误。

AI 工程中链表不如数组常见，但在 LRU、内存块管理、调度队列等场景仍有价值。
