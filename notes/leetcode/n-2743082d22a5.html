211. 添加与搜索单词-数据结构设计

力扣原题

简短题意

实现 WordDictionary：addWord 添加小写单词，search 判断是否存在与查询匹配的完整单词。查询中的 . 匹配任意一个字母，不能匹配空串或多个字符。

思路解析

Trie 存所有单词，末端 isEnd 标识完整单词。普通查询字母只走对应孩子，遇到 . 则递归尝试所有存在的孩子。走完查询长度时，必须检查 isEnd。节点用 vector 管理，下标连接，避免手动管理节点释放。

C++ 主模板

class WordDictionary {
    // 普通字符走唯一孩子，点号尝试所有存在的孩子。
    struct TrieNode {
        array<int,26> child;
        bool isEnd = false;
        TrieNode() { child.fill(-1); }
    };
    vector<TrieNode> nodes;
    bool match(const string& word, int pos, int node) {
        if (pos == static_cast<int>(word.size())) return nodes[node].isEnd;
        if (word[pos] != '.') {
            int next = nodes[node].child[word[pos] - 'a'];
            return next != -1 && match(word, pos + 1, next);
        }
        for (int next : nodes[node].child) {
            if (next != -1 && match(word, pos + 1, next)) return true;
        }
        return false;
    }
public:
    WordDictionary() : nodes(1) {}
    void addWord(string word) {
        int node = 0;
        for (char c : word) {
            int letter = c - 'a';
            if (nodes[node].child[letter] == -1) {
                int next = nodes.size();
                nodes[node].child[letter] = next;
                nodes.emplace_back();
            }
            node = nodes[node].child[letter];
        }
        nodes[node].isEnd = true;
    }
    bool search(string word) { return match(word, 0, 0); }
};

示例与复杂度

添加 bad、dad、mad 后，search(".ad") 为 true，search("b..") 为 true，search("ba") 为 false。

添加长度 L 的词 O(L)；无通配查询 O(L)，通配查询最坏遍历相关 Trie 节点 O(T)，T 为节点总数。存储 O(T)，查询递归栈 O(L)。

易错点与一类题总结

前缀存在不代表完整词存在；vector 扩容可能使引用失效，模板只保存节点下标，不持有节点引用跨 emplace_back。array 需 。

前缀树上普通字符沿唯一边走，通配字符分支搜索；递归终点按完整单词标记判定。

212. 单词搜索II

力扣原题

简短题意

给定小写字母网格 board 和不同单词组成的 words，返回能在网格中拼出的全部单词。路径按上下左右相邻格移动，一个单词路径中不能重复使用同一格；同一单词只返回一次。

思路解析

把全部单词建成 Trie，从每个格子做 DFS，沿网格路径同时沿 Trie 前缀走。没有对应 Trie 边就停止，因为没有任何目标词以当前路径为前缀。到达单词末端就收集，并清空末端编号避免重复输出。当前格临时标 #，递归返回后恢复，保证路径不重用而不同路径仍能用。

C++ 主模板

class Solution {
    struct TrieNode {
        array<int,26> child;
        int wordIndex = -1;
        TrieNode() { child.fill(-1); }
    };
    vector<TrieNode> trie;
    vector<string> answer;
    int rows, cols;
    void dfs(vector<vector<char>>& board, int r, int c, int node, vector<string>& words) {
        char letter = board[r][c];
        if (letter == '#') return;
        int next = trie[node].child[letter - 'a'];
        if (next == -1) return;
        if (trie[next].wordIndex != -1) {
            answer.push_back(words[trie[next].wordIndex]);
            trie[next].wordIndex = -1; // 同词的多条路径只输出一次
        }
        board[r][c] = '#';
        int dr[4] = {-1,1,0,0}, dc[4] = {0,0,-1,1};
        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < rows && nc >= 0 && nc < cols) dfs(board,nr,nc,next,words);
        }
        board[r][c] = letter;
    }
public:
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        trie.clear(); trie.emplace_back(); answer.clear();
        for (int i = 0; i < static_cast<int>(words.size()); i++) {
            int node = 0;
            for (char c : words[i]) {
                int letter = c - 'a';
                if (trie[node].child[letter] == -1) {
                    int next = trie.size(); trie[node].child[letter] = next;
                    trie.emplace_back();
                }
                node = trie[node].child[letter];
            }
            trie[node].wordIndex = i;
        }
        rows = board.size(); cols = board[0].size();
        for (int r = 0; r < rows; r++) for (int c = 0; c < cols; c++) dfs(board,r,c,0,words);
        return answer;
    }
};

示例与复杂度

经典 4×4 网格中 words=[oath,pea,eat,rain]，能找到 oath、eat；沿 oath 的路径返回后，格子恢复供后续词搜索。

建树 O(S)，S 为所有单词总长度。搜索宽松上界 O(mn×4^L)，L 为最大词长，实际由 Trie 前缀限制；额外空间 O(S+L)，输出另计。

易错点与一类题总结

发现短词后不能停止搜索，仍可能存在以它为前缀的长词；临时标记必须恢复；Word Search I 的逐词搜索在词表很大时重复扫描多。

多目标串同时搜索，用 Trie 共享前缀，将网格 DFS 的状态扩展为“格子位置 + 当前前缀节点”。
